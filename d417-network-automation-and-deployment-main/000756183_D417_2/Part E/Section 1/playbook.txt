---
- name: Create a user account in each AWS server instance
  hosts: 
   - ubuntu_srv
   - redhat_srv
  become: true
  vars:
    gather_facts: true

  tasks:
   - name: Add user account
     ansible.builtin.user:
       name: AdminWGU
       state: present
       password: "WGU123"

- name: Give AdminWGU root access on Ubuntu
  hosts: ubuntu_srv
  become: true
  tasks:
    - name: Add AdminWGU to sudo group
      ansible.builtin.user:
        name: AdminWGU
        groups: sudo
        append: true

- name: Give AdminWGU root access on Red Hat
  hosts: redhat_srv
  become: true
  tasks:
    - name: Add AdminWGU to wheel group
      ansible.builtin.user:
        name: AdminWGU
        groups: wheel
        append: true
